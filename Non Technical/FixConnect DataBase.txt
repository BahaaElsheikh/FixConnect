CREATE DATABASE SWE_PROJECT;
GO

USE SWE_PROJECT;
GO

-- ======================
-- 1. User
-- ======================
CREATE TABLE User (
    User_ID INT IDENTITY(1,1) PRIMARY KEY,
    Full_Name VARCHAR(100) NOT NULL,
    Email VARCHAR(100) UNIQUE NOT NULL,
    Password VARCHAR(255) NOT NULL, -- Added 
    PasswordHash VARCHAR(255) NOT NULL,
    Phone CHAR(11) NOT NULL,
    Role_Type INT NOT NULL,
    Created_At DATETIME DEFAULT GETDATE()
);

-- ======================
-- 2. Customer (TPT)
-- ======================
CREATE TABLE Customer (
    User_ID INT PRIMARY KEY,
    Address VARCHAR(100),
    Total_Requests INT DEFAULT 0,

    FOREIGN KEY (User_ID) REFERENCES User(User_ID)
        ON DELETE CASCADE
);

-- ======================
-- 3. Worker (TPT)
-- ======================
CREATE TABLE Worker (
    User_ID INT PRIMARY KEY,
    Specialty VARCHAR(100),
    Bio VARCHAR(255),
    Is_Verified BIT DEFAULT 0,
    Avg_Rating DECIMAL(3,2) DEFAULT 0,
    Availability_Status INT,

    FOREIGN KEY (User_ID) REFERENCES User(User_ID)
        ON DELETE CASCADE
);

-- ======================
-- 4. Admin (TPT)
-- ======================
CREATE TABLE Admins (
    User_ID INT PRIMARY KEY,
    Permissions_Level INT,

    FOREIGN KEY (User_ID) REFERENCES User(User_ID)
        ON DELETE CASCADE
);

-- ======================
-- 5. Regions
-- ======================
CREATE TABLE Region (
    Region_ID INT IDENTITY(1,1) PRIMARY KEY,
    Region_Name VARCHAR(100) UNIQUE NOT NULL
);

-- ======================
-- 6. Worker Regions (Many-to-Many)
-- ======================
CREATE TABLE Works_At (
    User_ID INT,
    Region_ID INT,

    PRIMARY KEY (User_ID, Region_ID),

    FOREIGN KEY (User_ID) REFERENCES Worker(User_ID)
        ON DELETE CASCADE,

    FOREIGN KEY (Region_ID) REFERENCES Region(Region_ID)
        ON DELETE CASCADE
);

-- ======================
-- 7. Requests
-- ======================
CREATE TABLE Request (
    Request_ID INT IDENTITY(1,1) PRIMARY KEY,
    Title VARCHAR(100) NOT NULL,
    Description VARCHAR(300),
    Request_Type INT,
    

    Target_Worker_ID INT NULL,-- NullAble

    Status INT NOT NULL,
    User_ID INT NOT NULL, -- Customer
    Region_ID INT NOT NULL,
    Created_At DATETIME DEFAULT GETDATE(),

    FOREIGN KEY (User_ID) REFERENCES Customer(User_ID)
        ON DELETE CASCADE,

    FOREIGN KEY (Region_ID) REFERENCES Region(Region_ID),

    FOREIGN KEY (Target_Worker_ID) REFERENCES Worker(User_ID)
);

-- ======================
-- 8. Proposal
-- ======================
CREATE TABLE Proposal (
    Proposal_ID INT IDENTITY(1,1) PRIMARY KEY,
    Labor_Cost DECIMAL(10,2),
    Material_Cost DECIMAL(10,2),
    Duration_Estimate INT, -- days 
    Status INT,

    User_ID INT NOT NULL, -- Customer
    Request_ID INT NOT NULL,
    Worker_ID INT NOT NULL,

    FOREIGN KEY (User_ID) REFERENCES Customer(User_ID),

    FOREIGN KEY (Request_ID) REFERENCES Request(Request_ID)
        ON DELETE CASCADE,

    FOREIGN KEY (Worker_ID) REFERENCES Worker(User_ID)
);

-- ======================
-- 9. Job
-- ======================
CREATE TABLE Job (
    Job_ID INT IDENTITY(1,1) PRIMARY KEY,
    Live_Invoice_Total DECIMAL(10,2),
    Start_Date DATE,
    End_Date DATE,
    Status INT,

    Proposal_ID INT UNIQUE,

    FOREIGN KEY (Proposal_ID) REFERENCES Proposal(Proposal_ID)
        ON DELETE CASCADE
);

-- ======================
-- 10. Portfolio
-- ======================
CREATE TABLE Portfolio_Item (
    Item_ID INT IDENTITY(1,1) PRIMARY KEY,
    Title VARCHAR(100),
    Description VARCHAR(300),
    Image_URL VARCHAR(255),

    User_ID INT,

    FOREIGN KEY (User_ID) REFERENCES Worker(User_ID)
        ON DELETE CASCADE
);

-- ======================
-- 11. Review
-- ======================
CREATE TABLE Review (
    Review_ID INT IDENTITY(1,1) PRIMARY KEY,
    Rating_Value INT CHECK (Rating_Value BETWEEN 1 AND 5),
    Comment VARCHAR(300),

    User_ID INT, -- Customer
    Job_ID INT UNIQUE,
    Worker_ID INT,

    FOREIGN KEY (User_ID) REFERENCES Customer(User_ID),
    FOREIGN KEY (Job_ID) REFERENCES Job(Job_ID),
    FOREIGN KEY (Worker_ID) REFERENCES Worker(User_ID)
);

-- ======================
-- 12. Wallet
-- ======================
CREATE TABLE Wallet (
    Wallet_ID INT IDENTITY(1,1) PRIMARY KEY,
    Worker_ID INT UNIQUE,
    Balance DECIMAL(10,2) DEFAULT 0,

    FOREIGN KEY (Worker_ID) REFERENCES Worker(User_ID)
        ON DELETE CASCADE
);

-- ======================
-- 13. Transactions
-- ======================
CREATE TABLE Transactions (
    Transaction_ID INT IDENTITY(1,1) PRIMARY KEY,
    Amount DECIMAL(10,2),
    Type_ INT,
    Created_At DATETIME DEFAULT GETDATE(),

    Wallet_ID INT,

    FOREIGN KEY (Wallet_ID) REFERENCES Wallet(Wallet_ID)
);
